"""Online value/critical-service scheduler, standard-library JSONL.

The bounded calendar propagates valuable descendant priorities, predicts running
attempt residuals from the public duration prior, and reserves worker and shared
memory intervals. Contiguous equal-cache reservations avoid setup. Only starts
at the current tick are emitted; every observation causes replanning. The fast
cache dispatcher uses deadline feasibility, remaining path cost, critical-class
emphasis, and heterogeneous worker speed. No future workload data is accessed.
"""
import sys,json,math
class Dispatcher:
    def __init__(self,mode='density',power=1.25,urgency=2.0,mean=1.15,mempen=0.35,wait=0.8,**kw):
        self.mode=mode; self.power=power; self.urgency=urgency; self.mean=mean; self.mempen=mempen; self.wait=wait
    def choose(self,s):
        if not s['ready'] or not s['idle_workers']: return []
        now=s['now']; cold=s['cold_start']; jobs={j['id']:j for j in s['jobs']}; done=s['completed']; workers=s['workers']
        live={i:j for i,j in jobs.items() if i not in done}; children={i:[] for i in live}
        for i,j in live.items():
            for d in j['deps']:
                if d in children: children[d].append(i)
        dur={i:min(e*self.mean+cold for k,e in j['estimate'].items()) for i,j in live.items()}
        # Iterative topological order, robust to input ordering and deep DAGs.
        deg={i:sum(d in live for d in j['deps']) for i,j in live.items()}; q=[i for i in live if not deg[i]]; order=[]
        for i in q:
            order.append(i)
            for c in children[i]:
                deg[c]-=1
                if not deg[c]: q.append(c)
        info={}
        for i in reversed(order):
            j=live[i]; v=(j['weight']**self.power if j['weight'] else 0); dl=j['deadline']; rem=dur[i]
            opts=[(v,dl,rem)] if v else []
            for c in children[i]:
                cv,cd,cr=info[c]; opts.append((cv,cd,cr+dur[i]))
            if opts:
                v,dl,rem=max(opts,key=lambda x:x[0]/max(1,x[2]))
                if len(opts)>1: v=max(v,sum(x[0] for x in opts)*0.7)
            else: v=.08; dl=j['deadline']; rem=dur[i]
            info[i]=(v,dl,rem)
        running={r['worker_id']:r for r in s['running']}; avail={}
        for w in workers:
            if not w['online']: continue
            if w['id'] in running:
                r=running[w['id']]; j=jobs[r['job_id']]; expected=j['estimate'][w['kind']]*self.mean+(cold if w['last_cache']!=j['cache'] else 0)
                avail[w['id']]=max(1,expected-(now-r['start']))
            else: avail[w['id']]=0
        pairs=[]
        for j in s['ready']:
            i=j['id']; v,dl,rem=info[i]
            best=min((avail[w['id']]+j['estimate'][w['kind']]*self.mean+(cold if w['last_cache']!=j['cache'] else 0) for w in workers if w['id'] in avail and w['kind'] in j['estimate'] and w['memory']>=j['memory']),default=1e9)
            for w in s['idle_workers']:
                if w['kind'] not in j['estimate'] or j['memory']>w['memory'] or j['memory']>s['memory_free']: continue
                p=j['estimate'][w['kind']]*self.mean+(cold if w['last_cache']!=j['cache'] else 0); remain=rem-dur[i]+p; slack=dl-now-remain
                if self.mode=='edf':
                    score=(100000 if v>=16**self.power else 10000 if v>=4**self.power else 1000)- (dl-remain*.6)
                    if dl<now: score=-100000+(v/rem)
                elif self.mode=='density':
                    # Smooth probability of feasibility, and ATC urgency among feasible work.
                    spread=max(3,.25*remain)
                    prob=1/(1+math.exp(max(-50,min(50,-slack/spread))))
                    urgent=math.exp(-max(0,slack)/(self.urgency*max(10,rem)))
                    score=v/(max(1,rem)**.7 * max(1,p)**.3)*prob*(.15+.85*urgent)
                    score/=1+self.mempen*j['memory']*len(workers)/s['memory']
                    if dl<=now: score=1e-7*v/max(1,rem)
                else:
                    score=v/(max(1,remain))*max(0.001, min(1,(dl-now)/max(1,remain)))
                    if dl<=now: score*=1e-6
                if p>best+max(2,best*(1-self.wait)) and self.mode!='edf': score*= (best/p)**3
                pairs.append((score,-p,i,w['id'],j['memory']))
        pairs.sort(reverse=True); used=set(); uw=set(); mem=s['memory_free']; out=[]
        for score,negp,i,w,m in pairs:
            if i in used or w in uw or m>mem: continue
            out.append(dict(job_id=i,worker_id=w)); used.add(i); uw.add(w); mem-=m
        return out

"""Deadline/value ranked DAG list scheduling with virtual worker and memory reservations."""
import sys,json,math,bisect,heapq
class Calendar:
    def __init__(self,power=1.5,mean=1.15,urgency=1.2,mempen=.5,exponent=.7,limit=40,lookahead=250,tail=.01,**kw):
        self.power=power; self.mean=mean; self.urgency=urgency; self.mempen=mempen; self.exponent=exponent; self.limit=limit; self.lookahead=lookahead; self.tail=tail
    def choose(self,s):
        if not s['ready'] or not s['idle_workers']: return []
        if not any(j['memory']<=min(s['memory_free'],w['memory']) and w['kind'] in j['estimate'] for j in s['ready'] for w in s['idle_workers']): return []
        if len(s['ready'])==1 and not s['running']:
            j=s['ready'][0]
            opts=[(j['estimate'][w['kind']]*self.mean+(s['cold_start'] if w['last_cache']!=j['cache'] else 0),k,w) for k,w in enumerate(s['idle_workers']) if w['kind'] in j['estimate'] and j['memory']<=min(w['memory'],s['memory_free'])]
            if opts:
                w=min(opts,key=lambda x:(x[0],x[1]))[2]; return [dict(job_id=j['id'],worker_id=w['id'])]
        now=s['now']; cold=s['cold_start']; horizon=s['horizon']; done=s['completed']; jobs={j['id']:j for j in s['jobs']}; active={r['job_id']:r for r in s['running']}
        ws=[w for w in s['workers'] if w['online']]; widx={w['id']:k for k,w in enumerate(ws)}; cal=[[] for w in ws]; intervals=[]; finishes={i:now for i in done}; planned={}
        for i,r in active.items():
            w=ws[widx[r['worker_id']]]; j=jobs[i]; elapsed=now-r['start']; setup=cold if w['last_cache']!=j['cache'] else 0
            ds=[max(1,int(j['estimate'][w['kind']]*x))+setup for x in (.7,1,1,1.4,1.8)]; ds=[d for d in ds if d>elapsed]
            p=max(1,(sum(ds)/len(ds) if ds else elapsed+max(1,j['estimate'][w['kind']]*.3))-elapsed)
            end=now+p; cal[widx[w['id']]].append((now,end,i)); intervals.append((now,end,j['memory'])); finishes[i]=end
        live={i:j for i,j in jobs.items() if i not in done}; children={i:[] for i in live}; deg={i:0 for i in live}
        for i,j in live.items():
            for d in j['deps']:
                if d in live: children[d].append(i); deg[i]+=1
        order=[i for i in live if not deg[i]]
        for i in order:
            for c in children[i]:
                deg[c]-=1
                if not deg[c]: order.append(c)
        minp={i:min(j['estimate'].values())*self.mean+cold for i,j in live.items()}; info={}
        for i in reversed(order):
            j=jobs[i]; opts=[(j['weight']**self.power,j['deadline'],minp[i])] if j['weight'] else []
            for c in children[i]:
                v,dl,rem=info[c]; opts.append((v,dl,rem+minp[i]))
            info[i]=max(opts,key=lambda x:x[0]/max(1,x[2]),default=(.06,j['deadline'],minp[i]))
        def priority(i):
            v,dl,rem=info[i]; slack=dl-now-rem; prob=1/(1+math.exp(max(-50,min(50,-slack/max(3,.25*rem)))))
            score=v/max(1,rem)**self.exponent*(self.tail+(1-self.tail)*prob)*(.15+.85*math.exp(-max(0,slack)/(self.urgency*max(10,rem))))
            if dl<=now: score=1e-7*v/max(1,rem)
            return score
        deg={i:sum(d not in finishes for d in j['deps']) for i,j in live.items() if i not in finishes}; heap=[(-priority(i),i) for i in deg if not deg[i]]; heapq.heapify(heap)
        def memslot(t,p,m,upper):
            for _ in range(len(intervals)+2):
                if t+p>upper: return None
                used=0; ev=[]
                for a,b,v in intervals:
                    if a<=t<b: used+=v; ev.append((b,-v))
                    elif t<a<t+p: ev.append((a,v)); ev.append((b,-v))
                if used+m>s['memory']:
                    t=min(b for a,b,v in intervals if a<=t<b); continue
                ev.sort(); conflict=None; pos=0
                while pos<len(ev):
                    tm=ev[pos][0]
                    if tm>=t+p: break
                    while pos<len(ev) and ev[pos][0]==tm: used+=ev[pos][1]; pos+=1
                    if used+m>s['memory']: conflict=tm; break
                if conflict is None: return t
                t=min(b for a,b,v in intervals if a<=conflict<b)
            return None
        steps=0; mem=s['memory_free']; ready={j['id'] for j in s['ready']}; out=[]; idleids={w['id'] for w in s['idle_workers']}
        while heap and steps<self.limit:
            neg,i=heapq.heappop(heap); j=jobs[i]; steps+=1
            earliest=max([now,*[finishes[d] for d in j['deps']]])
            if earliest>now+self.lookahead: continue
            best=None
            for wi,w in enumerate(ws):
                if w['kind'] not in j['estimate'] or w['memory']<j['memory']: continue
                prev=now; prevcache=w['last_cache']
                for a,b,ji in [*cal[wi],(horizon,horizon,None)]:
                    t=max(earliest,prev); warm=t==prev and prevcache==j['cache']; p=j['estimate'][w['kind']]*self.mean+(0 if warm else cold)
                    upper=a
                    if best is not None: upper=min(upper,best[0])
                    if t+p<=upper:
                        t=memslot(t,p,j['memory'],upper)
                        if t is not None and warm and t!=prev:
                            p+=cold; t=memslot(t,p,j['memory'],upper)
                        if t is not None:
                            cand=(t+p,t,wi)
                            if best is None or cand<best: best=cand
                    prev=max(prev,b)
                    if ji is not None: prevcache=jobs[ji]['cache']
                    if best is not None and prev>=best[0]: break
            if best is None: continue
            end,t,wi=best; bisect.insort(cal[wi],(t,end,i)); intervals.append((t,end,j['memory'])); finishes[i]=end
            if t==now and i in ready:
                out.append(dict(job_id=i,worker_id=ws[wi]['id'])); mem-=j['memory']; idleids.discard(ws[wi]['id'])
            for c in children[i]:
                if c in deg:
                    deg[c]-=1
                    if not deg[c]: heapq.heappush(heap,(-priority(c),c))
            if not any(jobs[x]['memory']<=min(mem,w['memory']) and w['kind'] in jobs[x]['estimate'] for x in ready if x not in finishes for w in ws if w['id'] in idleids): break
        return out


class Scheduler:
    """Use immediate dispatch when cache reuse dominates and memory is ample.
    Otherwise protect valuable dependency paths with a bounded calendar plan.
    All features are computed from the released observation only.
    """
    def __init__(self):
        self.dispatch=Dispatcher()
        self.calendar=Calendar()
    def choose(self,s):
        if not s['ready'] or not s['idle_workers']: return []
        jobs=s['jobs']; middle=len(jobs)//2
        typical_memory=sorted(j['memory'] for j in jobs)[middle]
        typical_duration=sorted(min(j['estimate'].values()) for j in jobs)[middle]
        plentiful=s['memory']>=len(s['workers'])*typical_memory*1.1
        if plentiful and s['cold_start']>=.6*typical_duration:
            return self.dispatch.choose(s)
        return self.calendar.choose(s)

if __name__=='__main__':
    policy=Scheduler()
    for line in sys.stdin:
        request=json.loads(line)
        actions=policy.choose(request['state'])
        print(json.dumps({'request_id':request['request_id'],'assignments':actions},separators=(',',':')),flush=True)
